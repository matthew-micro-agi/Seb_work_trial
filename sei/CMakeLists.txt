cmake_minimum_required(VERSION 3.16)
project(sei CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(sei STATIC src/sei.cpp)
target_include_directories(sei PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/../kmatch/include)
target_compile_options(sei PRIVATE -Wall -Wextra)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(test_sei tests/test_sei.cpp)
  target_link_libraries(test_sei sei)
  add_test(NAME test_sei COMMAND test_sei ${CMAKE_CURRENT_SOURCE_DIR}/tests/vectors.txt)
  add_test(NAME test_sei_py COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sei.py)
  add_test(NAME test_faults_py COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_faults.py)
endif()
