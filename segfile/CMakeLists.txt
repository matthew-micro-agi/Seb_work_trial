cmake_minimum_required(VERSION 3.16)
project(segfile CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)

if(NOT TARGET sei)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../sei sei)
endif()
if(NOT TARGET kmatch)
  set(KMATCH ${CMAKE_CURRENT_SOURCE_DIR}/../kmatch)
  add_library(kmatch STATIC ${KMATCH}/src/types.cpp ${KMATCH}/src/edges.cpp ${KMATCH}/src/matcher.cpp)
  target_include_directories(kmatch PUBLIC ${KMATCH}/include)
endif()

add_library(segfile STATIC src/segfile.cpp)
target_include_directories(segfile PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(segfile PUBLIC sei Threads::Threads)
target_compile_options(segfile PRIVATE -Wall -Wextra)

add_executable(kpipe-synth tools/synth.cpp)
target_link_libraries(kpipe-synth segfile kmatch)
add_executable(ringfsck tools/ringfsck.cpp)
target_link_libraries(ringfsck segfile kmatch)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(test_segfile tests/test_segfile.cpp)
  target_link_libraries(test_segfile segfile)
  add_test(NAME test_segfile COMMAND test_segfile)
  add_test(NAME test_synth_fixture COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture.sh $<TARGET_FILE:kpipe-synth> $<TARGET_FILE:ringfsck>)
endif()
